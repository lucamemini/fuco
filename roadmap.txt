# FUCO - Piano di Miglioramento Completo

## 🎯 Executive Summary

FUCO è ben strutturato ma ha margini significativi di miglioramento in:
- **Performance**: Cache in-memory volatile, nessun connection pooling
- **Sicurezza**: Mancano rate limiting, CSRF, input validation approfondita
- **Scalabilità**: Polling sincrono blocca worker, manca queue system
- **Resilienza**: Nessun circuit breaker per Cortex failures

### 1. **Cache Persistente con Redis** ⭐⭐⭐ (DONE)
**Problema attuale**: Cache in-memory (`_report_cache`) si perde al restart.

**Implementazione**:
```python
# requirements.txt
redis==5.0.1
flask-caching==2.1.0

# config.py
REDIS_URL = os.getenv('REDIS_URL', 'redis://localhost:6379/0')
CACHE_TYPE = 'redis'

# utils.py
from flask_caching import Cache
from redis import Redis
import pickle

redis_client = Redis.from_url(config.REDIS_URL, decode_responses=False)

def get_cached_report(job_id):
    """Cache con Redis - persiste tra restart"""
    cache_key = f"report:{job_id}"
    
    # 1. Check cache
    cached = redis_client.get(cache_key)
    if cached:
        report = pickle.loads(cached)
        logger.debug(f"Report {job_id} da Redis cache")
        return report
    
    # 2. Fetch from Cortex
    report = cortex_api_call(cortex_api.jobs.get_report, job_id)
    
    # 3. Cache solo status finali
    if hasattr(report, 'status') and report.status in ("Success", "Failure"):
        ttl = config.CACHE_TTL_MINUTES * 60
        redis_client.setex(cache_key, ttl, pickle.dumps(report))
        logger.info(f"Report {job_id} cached in Redis (TTL: {ttl}s)")
    
    return report

def get_cache_stats():
    """Stats con Redis"""
    keys = redis_client.keys("report:*")
    stats = {
        'total_reports': len(keys),
        'memory_usage': redis_client.info('memory')['used_memory_human'],
        'by_status': {}
    }
    
    # Sample 100 reports per performance
    for key in keys[:100]:
        try:
            report = pickle.loads(redis_client.get(key))
            status = getattr(report, 'status', 'Unknown')
            stats['by_status'][status] = stats['by_status'].get(status, 0) + 1
        except:
            pass
    
    return stats
```

**Benefici**:
- ✅ Persistenza tra restart
- ✅ Clustering multi-istanza
- ✅ TTL automatico
- ✅ Monitoring integrato

### 2. **Task Queue con Celery** ⭐⭐⭐
**Problema attuale**: Polling sincrono blocca i worker Flask.

**Implementazione**:
```python
# requirements.txt
celery==5.3.4
celery[redis]==5.3.4

# celery_app.py
from celery import Celery

celery_app = Celery(
    'fuco',
    broker=config.REDIS_URL,
    backend=config.REDIS_URL
)

celery_app.conf.update(
    task_serializer='pickle',
    accept_content=['pickle', 'json'],
    result_serializer='pickle',
    task_track_started=True,
    task_time_limit=600,  # 10 min timeout
    worker_prefetch_multiplier=4,
)

@celery_app.task(bind=True, max_retries=3)
def poll_job_async(self, job_id, max_attempts=15):
    """Background polling task"""
    try:
        for attempt in range(max_attempts):
            time.sleep(3 + attempt * 0.5)
            report = cortex_api_call(cortex_api.jobs.get_report, job_id)
            
            if report.status in ("Success", "Failure"):
                # Cache automaticamente
                cache_key = f"report:{job_id}"
                redis_client.setex(
                    cache_key, 
                    config.CACHE_TTL_MINUTES * 60,
                    pickle.dumps(report)
                )
                return {'status': 'completed', 'report_status': report.status}
        
        return {'status': 'timeout', 'attempts': max_attempts}
        
    except Exception as e:
        logger.error(f"Polling error for {job_id}: {e}")
        raise self.retry(exc=e, countdown=5)

# routes.py - Uso asincrono
@routes_bp.route('/api/submit_job', methods=['POST'])
def api_submit_job():
    # Sottometti job a Cortex
    job_result = utils.run_analysis(analyzer, datatype, data, tlp, pap)
    job_id = job_result['id']
    
    # Avvia polling in background
    task = poll_job_async.delay(job_id)
    
    return jsonify({
        'status': 'queued',
        'job_id': job_id,
        'task_id': task.id,
        'poll_url': f'/api/task_status/{task.id}'
    })

@routes_bp.route('/api/task_status/<task_id>')
def task_status(task_id):
    """Check Celery task status"""
    task = celery_app.AsyncResult(task_id)
    
    if task.state == 'PENDING':
        return jsonify({'state': 'pending'})
    elif task.state == 'SUCCESS':
        return jsonify({'state': 'success', 'result': task.result})
    elif task.state == 'FAILURE':
        return jsonify({'state': 'failed', 'error': str(task.info)})
    else:
        return jsonify({'state': task.state, 'info': str(task.info)})
```

```python
# utils.py
from requests.adapters import HTTPAdapter
from urllib3.util.retry import Retry
import requests

class CortexAPIWrapper:
    """Wrapper con connection pooling"""
    
    def __init__(self, host, api_key):
        self.host = host
        self.api_key = api_key
        self.session = self._create_session()
        self.api = Api(host, api_key)
        # Inietta session nella API
        self.api._session = self.session
    
    def _create_session(self):
        session = requests.Session()
        
        # Retry strategy
        retry = Retry(
            total=3,
            backoff_factor=0.5,
            status_forcelist=[500, 502, 503, 504],
            allowed_methods=["GET", "POST", "PUT", "DELETE"]
        )
        
        # Adapter con pooling
        adapter = HTTPAdapter(
            pool_connections=20,
            pool_maxsize=50,
            max_retries=retry,
            pool_block=False
        )
        
        session.mount('http://', adapter)
        session.mount('https://', adapter)
        
        # Headers comuni
        session.headers.update({
            'Authorization': f'Bearer {self.api_key}',
            'Content-Type': 'application/json'
        })
        
        return session
    
    def close(self):
        """Cleanup"""
        self.session.close()

# Inizializzazione globale
cortex_wrapper = CortexAPIWrapper(cfg.cortex["host"], cfg.cortex["apikey"])
cortex_api = cortex_wrapper.api

# Cleanup on shutdown
import atexit
atexit.register(cortex_wrapper.close)
```

```python
# routes.py
from concurrent.futures import ThreadPoolExecutor, as_completed

@routes_bp.route('/api/short', methods=['POST'])
def api_short():
    # ... parsing input ...
    
    if len(input_items) > 1:
        # Batch processing parallelo
        final_results = batch_analyze(input_items, analyzer_list)
    else:
        # Single item
        final_results = sequential_analyze(input_items, analyzer_list)
    
    return jsonify(final_results)

def batch_analyze(input_items, analyzer_list, max_workers=10):
    """Analisi parallela per performance"""
    results = []
    
    with ThreadPoolExecutor(max_workers=max_workers) as executor:
        # Submit all jobs
        future_to_job = {}
        for data, dtype in input_items:
            for analyzer in analyzer_list:
                future = executor.submit(
                    utils.run_analysis, 
                    analyzer, dtype, data
                )
                future_to_job[future] = (data, analyzer)
        
        # Collect job IDs
        job_results = []
        for future in as_completed(future_to_job):
            data, analyzer = future_to_job[future]
            try:
                job = future.result(timeout=10)
                job_results.append((data, analyzer, job))
            except Exception as e:
                logger.error(f"Job submission failed: {e}")
        
        # Parallel polling
        poll_futures = {}
        for data, analyzer, job in job_results:
            future = executor.submit(
                utils.poll_job, 
                job['id'],
                config.API_SHORT_MAX_ATTEMPTS,
                config.API_SHORT_INITIAL_DELAY
            )
            poll_futures[future] = (data, analyzer, job)
        
        # Collect results
        for future in as_completed(poll_futures):
            data, analyzer, job = poll_futures[future]
            try:
                report = future.result(timeout=120)
                results.append(format_result(data, analyzer, report))
            except Exception as e:
                logger.error(f"Polling failed: {e}")
    
    return results
```

```python
# models.py
import sqlite3
from datetime import datetime
import json

class SearchHistoryDB:
    def __init__(self, db_path='fuco_history.db'):
        self.db_path = db_path
        self._init_db()
    
    def _init_db(self):
        conn = sqlite3.connect(self.db_path)
        conn.execute('''
            CREATE TABLE IF NOT EXISTS searches (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                observable TEXT NOT NULL,
                datatype TEXT NOT NULL,
                analyzers TEXT NOT NULL,
                job_ids TEXT NOT NULL,
                timestamp DATETIME DEFAULT CURRENT_TIMESTAMP,
                status TEXT DEFAULT 'pending',
                user_ip TEXT
            )
        ''')
        conn.execute('''
            CREATE INDEX IF NOT EXISTS idx_observable 
            ON searches(observable, datatype)
        ''')
        conn.execute('''
            CREATE INDEX IF NOT EXISTS idx_timestamp 
            ON searches(timestamp DESC)
        ''')
        conn.commit()
        conn.close()
    
    def save_search(self, observable, datatype, analyzers, job_ids, user_ip=None):
        conn = sqlite3.connect(self.db_path)
        conn.execute('''
            INSERT INTO searches 
            (observable, datatype, analyzers, job_ids, user_ip)
            VALUES (?, ?, ?, ?, ?)
        ''', (
            observable,
            datatype,
            json.dumps(analyzers),
            json.dumps(job_ids),
            user_ip
        ))
        conn.commit()
        conn.close()
    
    def get_recent(self, limit=50):
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute('''
            SELECT DISTINCT observable, datatype, timestamp
            FROM searches
            WHERE status = 'completed'
            ORDER BY timestamp DESC
            LIMIT ?
        ''', (limit,))
        
        results = {}
        for row in cursor.fetchall():
            observable, datatype, timestamp = row
            if datatype not in results:
                results[datatype] = {}
            if observable not in results[datatype]:
                results[datatype][observable] = timestamp
        
        conn.close()
        return results

# utils.py

    except Exception as e:
        logger.error(f"DB error: {e}")
        # Fallback a Cortex
        return get_recent_searches_cortex()
```

```python
# utils.py
import bleach
import validators
from html import escape

class InputValidator:
    """Validazione e sanitizzazione input"""
    
    @staticmethod
    def sanitize_observable(data: str, max_length=500) -> str:
        """Sanitizza observable"""
        if not data:
            raise ValueError("Observable cannot be empty")
        
        # Remove dangerous chars
        data = bleach.clean(data, tags=[], strip=True)
        
        # Trim
        data = data.strip()[:max_length]
        
        # Remove null bytes
        data = data.replace('\x00', '')
        
        # Remove multiple spaces
        data = ' '.join(data.split())
        
        return data
    
    @staticmethod
    def validate_ip(ip: str) -> bool:
        """Valida IP con validators library"""
        try:
            return validators.ipv4(ip) or validators.ipv6(ip)
        except:
            return False
    
    @staticmethod
    def validate_domain(domain: str) -> bool:
        """Valida domain"""
        try:
            return validators.domain(domain)
        except:
            return False
    
    @staticmethod
    def validate_url(url: str) -> bool:
        """Valida URL"""
        try:
            return validators.url(url)
        except:
            return False
    
    @staticmethod
    def validate_hash(hash_str: str, hash_type: str) -> bool:
        """Valida hash (MD5, SHA256, etc.)"""
        patterns = {
            'md5': r'^[a-fA-F0-9]{32}$',
            'sha1': r'^[a-fA-F0-9]{40}$',
            'sha256': r'^[a-fA-F0-9]{64}$'
        }
        
        pattern = patterns.get(hash_type.lower())
        if not pattern:
            return False
        
        return bool(re.match(pattern, hash_str))
```

```python
# config.py
API_KEYS_ENABLED = os.getenv('FUCO_API_KEYS_ENABLED', 'true').lower() == 'true'

# models.py
class APIKeyManager:
    """Gestione API keys"""
    
    def __init__(self):
        self.db_path = 'fuco_apikeys.db'
        self._init_db()
    
    def _init_db(self):
        conn = sqlite3.connect(self.db_path)
        conn.execute('''
            CREATE TABLE IF NOT EXISTS api_keys (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                key TEXT UNIQUE NOT NULL,
                name TEXT NOT NULL,
                created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
                last_used DATETIME,
                is_active BOOLEAN DEFAULT 1,
                rate_limit INTEGER DEFAULT 100,
                allowed_ips TEXT
            )
        ''')
        conn.commit()
        conn.close()
    
    def validate_key(self, api_key: str, client_ip: str = None) -> bool:
        """Valida API key"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute('''
            SELECT is_active, allowed_ips 
            FROM api_keys 
            WHERE key = ?
        ''', (api_key,))
        
        row = cursor.fetchone()
        
        if not row:
            conn.close()
            return False
        
        is_active, allowed_ips_json = row
        
        if not is_active:
            conn.close()
            return False
        
        # Check IP whitelist
        if allowed_ips_json and client_ip:
            allowed_ips = json.loads(allowed_ips_json)
            if client_ip not in allowed_ips:
                conn.close()
                return False
        
        # Update last_used
        cursor.execute('''
            UPDATE api_keys 
            SET last_used = CURRENT_TIMESTAMP 
            WHERE key = ?
        ''', (api_key,))
        
        conn.commit()
        conn.close()
        return True
    
    def create_key(self, name: str, allowed_ips: list = None) -> str:
        """Crea nuova API key"""
        import secrets
        api_key = f"fuco_{secrets.token_urlsafe(32)}"
        
        conn = sqlite3.connect(self.db_path)
        conn.execute('''
            INSERT INTO api_keys (key, name, allowed_ips)
            VALUES (?, ?, ?)
        ''', (api_key, name, json.dumps(allowed_ips) if allowed_ips else None))
        conn.commit()
        conn.close()
        
        return api_key

# routes.py
api_key_manager = APIKeyManager()

def api_key_required(f):
    """Decorator per API authentication"""
    @wraps(f)
    def decorated(*args, **kwargs):
        if not config.API_KEYS_ENABLED:
            return f(*args, **kwargs)
        
        api_key = request.headers.get('X-API-Key')
        
        if not api_key:

        if not api_key_manager.validate_key(api_key, client_ip):
            return jsonify({'error': 'Invalid or inactive API key'}), 403
        
        return f(*args, **kwargs)
    
    return decorated
```
